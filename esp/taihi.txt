// ===== 設定 =====
const int DOT_SIZE = 3;
const int SENSOR_PIN = 5;   // 受光部のADCピンに合わせて変更
const int T_MS = 200;       // Web側スライダーTと合わせる

const uint32_t BG_COLOR   = 0x000000U;
const uint32_t MAIN_COLOR = 0x00FF00U;

// ===== 受信結果 =====
String receivedSsid = "";
String receivedPass = "";
String currentBits = "";

enum ReceiveMode {
  WAIT_START,
  READ_SSID,
  READ_PASSWORD,
  FINISHED
};

ReceiveMode mode = WAIT_START;

// ===== 光の状態検出用 =====
bool lastLight = false;
unsigned long lastEdgeTime = 0;
unsigned long greenDuration = 0;
unsigned long blackDuration = 0;

// しきい値。まずはSerialで値を見て調整
int threshold = 2000;

// --------------------------------
// 画面描画
// --------------------------------

void drawQR() {
  int qrSize = sizeof(qr) / sizeof(qr[0]);

  int offsetX = 0;
  int offsetY = 0;

  for (int y = 0; y < qrSize; y++) {
    for (int x = 0; x < qrSize; x++) {
      if (qr[y][x]) {
        lcd.fillRect(
          offsetX + x * DOT_SIZE,
          offsetY + y * DOT_SIZE,
          DOT_SIZE,
          DOT_SIZE,
          MAIN_COLOR
        );
      }
    }
  }
}

void drawStaticScreen() {
  lcd.fillScreen(BG_COLOR);

  // 左上にQR
  drawQR();

  // 右側に説明
  lcd.setTextColor(MAIN_COLOR, BG_COLOR);

  int x = 115;

  lcd.setCursor(x, y);

void updateReceivedArea() {
  // 表示欄だけ黒で消す
  lcd.fillRect(55, 160, 260, 70, BG_COLOR);

  lcd.setTextColor(MAIN_COLOR, BG_COLOR);
  lcd.setTextSize(1);

  lcd.setCursor(55, 165);
  lcd.print(receivedSsid);

  lcd.setCursor(55, 195);
  lcd.print(receivedPass);

  lcd.setCursor(8, 220);
  lcd.print("bits:");
  lcd.print(currentBits);
  lcd.print("        ");
}

// --------------------------------
// 受信デコード
// --------------------------------

bool nearUnit(int value, int target) {
  // Tの何倍かをざっくり判定
  // 許容幅は広め。あとで調整してください。
  return abs(value - target) <= 1;
}

int toUnit(unsigned long durationMs) {
  return (durationMs + T_MS / 2) / T_MS;
}

void resetReceive() {
  receivedSsid = "";
  receivedPass = "";
  currentBits = "";
  mode = WAIT_START;
  updateReceivedArea();
}

void addBit(int bit) {
  currentBits += bit ? "1" : "0";

  // 8bitたまったら1文字に変換
  if (currentBits.length() >= 8) {
    int code = 0;

    for (int i = 0; i < 8; i++) {
      code <<= 1;
      if (currentBits[i] == '1') {
        code |= 1;
      }
    }

    // UTF-8 1バイト、つまりASCIIだけ採用
    if (code <= 0x7F) {
      if (mode == READ_SSID) {
        receivedSsid += char(code);
      } else if (mode == READ_PASSWORD) {
        receivedPass += char(code);
      }
    }

    currentBits = "";
    updateReceivedArea();
  }
}

void handlePulse(unsigned long gMs, unsigned long bMs) {
  int g = toUnit(gMs);
  int b = toUnit(bMs);

  Serial.print("G=");
  Serial.print(g);
  Serial.print(" B=");
  Serial.println(b);

  // 開始信号の後半部分っぽいものを見る
  // 今回は簡易版として G4 B1 を開始後の区切りとして使う
  if (mode == WAIT_START) {
    if (nearUnit(g, 4) && nearUnit(b, 1)) {
      mode = READ_SSID;
      receivedSsid = "";
      receivedPass = "";
      currentBits = "";
      updateReceivedArea();
    }
    return;
  }

  // G4 B1 は password header または end
  if (nearUnit(g, 4) && nearUnit(b, 1)) {
    if (mode == READ_SSID) {
      mode = READ_PASSWORD;
      currentBits = "";
    } else if (mode == READ_PASSWORD) {
      mode = FINISHED;
      currentBits = "";
    }

    updateReceivedArea();
    return;
  }

  // 0: green T, black T
  if (nearUnit(g, 1) && nearUnit(b, 1)) {
    addBit(0);
    return;
  }

  // 1: green T, black 3T
  if (nearUnit(g, 1) && nearUnit(b, 3)) {
    addBit(1);
    return;
  }
}

void readSensor() {
  int value = analogRead(SENSOR_PIN);
  bool light = value > threshold;

  unsigned long now = millis();

  // 光状態が変化した瞬間だけ処理
  if (light != lastLight) {
    unsigned long duration = now - lastEdgeTime;

    if (lastLight) {
      // 直前まで光っていた → green時間
      greenDuration = duration;
    } else {
      // 直前まで暗かった → black時間
      blackDuration = duration;

      // greenとblackが1セットそろったので判定
      handlePulse(greenDuration, blackDuration);
    }

    lastEdgeTime = now;
    lastLight = light;
  }
}

// --------------------------------
// setup / loop
// --------------------------------

void setup() {
  Serial.begin(115200);

  lcd.init();
  lcd.setRotation(3);
  lcd.setBrightness(128);

  drawStaticScreen();
  updateReceivedArea();

  lastLight = analogRead(SENSOR_PIN) > threshold;
  lastEdgeTime = millis();
}

void loop() {
  readSensor();
}
